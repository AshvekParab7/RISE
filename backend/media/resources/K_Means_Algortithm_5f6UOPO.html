# Scatter Plot
plt.figure(figsize=(8,6))

plt.scatter(
    data['Study_Hours'],
    data['Marks'],
    c=data['Cluster'],
    cmap='rainbow',
    s=80
)

# Plot cluster centroids
plt.scatter(
    kmeans.cluster_centers_[:,0],
    kmeans.cluster_centers_[:,1],
    color='black',
    marker='X',
    s=250,
    label='Centroids'
)

# Labels and Title
plt.xlabel("Study Hours")
plt.ylabel("Marks")
plt.title("K-Means Clustering of Students")

# Show legend
plt.legend()

# Show grid
plt.grid(True)

# Display plot
plt.show()
